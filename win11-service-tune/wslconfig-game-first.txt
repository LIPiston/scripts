# Template for C:\Users\<user>\.wslconfig  -- "game first" profile.
# Applied 2026-09-28 on a 16GB machine: WSL2 gives memory back to Windows while gaming.
# Restart WSL for changes to take effect:  wsl --shutdown
# Verify inside WSL: nproc ; grep MemTotal /proc/meminfo

[wsl2]
# Host has 8C/16T. 6 leaves headroom for the game; raise to 8 if you compile a lot.
processors=6
# Was 4.9GB (default is 50% of host RAM). 3GB = even more headroom for games.
memory=4GB
swap=4GB
# Idle VM shuts down after 2 min, releasing its memory (ms).
vmIdleTimeout=120000

[experimental]
# Return idle Linux page cache to Windows every few seconds.
# Measured: 1.19GB of guest cache was handed back within 30s (Vmmem 1781MB -> 686MB).
autoMemoryReclaim=gradual
